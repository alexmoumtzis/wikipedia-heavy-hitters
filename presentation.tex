\section{Motivation}
%================================================

\begin{frame}{The heavy-hitters problem}
  \begin{block}{Definition}
    Given a stream of total weight $N=\sum_i f(i)$ and a support $\varphi$, an item $x$ is a
    \alert{true \hh} iff its frequency $f(x) \ge \varphi N$.
  \end{block}
  \begin{itemize}
    \item One of the oldest and most-used primitives in streaming analytics
          \cite{cormodeFindingFrequentItems2008}.
    \item The alphabet of possible items $\gg$ available memory
          $\Rightarrow$ \textbf{exact counting is infeasible}.
    \item We need \alert{sub-linear space} approximations that still find the few
          items carrying most of the traffic.
  \end{itemize}
  \vspace{0.5em}
  \begin{exampleblock}{Running example}
    Which Wikipedia pages dominate the request traffic in a given snapshot?
    Access counts follow a strongly \textbf{Zipfian} distribution, perfect for
    heavy-hitter detection.
  \end{exampleblock}
\end{frame}

\begin{frame}{Project goal}
  \begin{alertblock}{We do \emph{not} try to crown a single ``best'' algorithm.}
    We characterise the \textbf{operational regimes} (memory budget, required
    error guarantee, data skew, and distribution model) under which each
    algorithmic \emph{family} becomes preferable.
  \end{alertblock}
  \vspace{0.5em}
  We compare two families along five axes:
  \begin{multicols}{2}
    \begin{itemize}
      \item Accuracy (precision / recall / error)
      \item Memory consumption
      \item Throughput
      \item Scalability across Spark partitions
      \item Distributed \textbf{mergeability}
    \end{itemize}
  \end{multicols}
  \vspace{0.3em}
  Reference points: two sampling summaries (Reservoir, Concise) and an
  \textbf{exact} Spark top-$k$ baseline.
\end{frame}

\section{Algorithm Families}
%================================================

\begin{frame}{Two families of summaries}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textcolor{sketchcol}{Sketch-based}}
        Fixed grid of \emph{hashed} counters; estimate the frequency of any queried item.
        \begin{itemize}
          \item \textbf{Count--Min (CMS)} \cite{cormodeImprovedDataStream2005}
          \item \textbf{Count Sketch / FastAMS} \cite{charikarFindingFrequentItems2002}
                (hashed AMS \cite{alonSpaceComplexityApproximating1999})
        \end{itemize}
        \textit{Linear} in the stream $\Rightarrow$ merge by summing tables.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{\textcolor{countercol}{Counter-based (insert-only)}}
        Bounded, \emph{explicit} set of (item, counter) pairs.
        \begin{itemize}
          \item \textbf{Misra--Gries (MG)} \cite{misraFindingRepeatedElements1982}
          \item \textbf{Space-Saving (SS)} \cite{metwallyEfficientComputationFrequent2005}
          \item \textbf{Lossy Counting (LC)} \cite{mankuApproximateFrequencyCounts2002}
        \end{itemize}
        Interpretable state; mergeable only with a \emph{lossy prune}
        \cite{agarwalMergeableSummaries2013}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Primer on the counter-based family (not in the course curriculum)

\begin{frame}{Counter-based algorithms: the shared idea}
  \begin{block}{One sentence}
    Keep an \textbf{explicit table of at most $k$ (item, counter) pairs}. On each
    arrival either update an entry we already track, or make room for the new one.
    We \emph{never} store more than $k$ items, so space is $\bigO{k}$ regardless of
    stream length.
  \end{block}
  \begin{itemize}
    \item The three algorithms differ only in \alert{what happens on a miss}
          (an item that is not currently tracked and the table is full).
    \item Because they hold real items, the output is directly interpretable:
          you read the heavy hitters straight off the table.
    \item Setting $k=\lceil 1/\varepsilon\rceil$ gives an additive error of at most
          $\varepsilon N$ on every reported count.
  \end{itemize}
  \centering\small
  \emph{Next: how Misra--Gries, Space-Saving, and Lossy Counting each resolve a miss.}
\end{frame}

\begin{frame}{Misra--Gries (MG): ``decrement everybody''}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Rule on each arrival of item $x$:}
      \begin{enumerate}
        \item If $x$ is tracked $\to$ increment its counter.
        \item Else if a slot is free $\to$ add $x$ with counter $1$.
        \item Else (table full) $\to$ \alert{decrement \emph{all} $k-1$ counters};
              drop any that hit $0$.
      \end{enumerate}
      \vspace{0.4em}
      Intuition: a decrement cancels one occurrence of $k-1$ \emph{different}
      items against one of $x$, so a truly frequent item survives the cancellations.
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Guarantee}
        Never \emph{over}-counts: $\hat f(x)\le f(x)\le \hat f(x)+\varepsilon N$.
        Any item with $f(x) > N/k$ is guaranteed to be kept.
      \end{block}
      \begin{exampleblock}{Why we care}
        MG summaries are \textbf{mergeable} (union the tables, then prune back to
        $k$), which drives the distributed part of this project.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Space-Saving (SS): ``overwrite the smallest''}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Rule on each arrival of item $x$:}
      \begin{enumerate}
        \item If $x$ is tracked $\to$ increment its counter.
        \item Else if a slot is free $\to$ add $x$ with counter $1$.
        \item Else (table full) $\to$ find the \alert{minimum} entry $e_{\min}$,
              \alert{replace its item with $x$}, and increment (keep its old count).
      \end{enumerate}
      \vspace{0.4em}
      The evicted count is inherited, so a new item's estimate carries an
      over-estimation term $\varepsilon_i$ (how much it may be inflated).
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Guarantee}
        Never \emph{under}-counts: $\text{count}_i-\varepsilon_i \le f(e_i) \le \text{count}_i$.
        Provably \emph{isomorphic} to MG, but it tracks an upper bound instead of a lower one.
      \end{block}
      \begin{exampleblock}{Trade-off}
        Great recall (nothing frequent is lost), but the inherited counts can
        inflate precision-critical estimates on tiny budgets.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Lossy Counting (LC): ``prune at bucket boundaries''}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Mechanism:}
      \begin{itemize}
        \item Split the stream into \emph{buckets} of width $w=\lceil 1/\varepsilon\rceil$.
        \item Track (item, count, $\Delta$) where $\Delta$ is the max possible
              undercount when the item was first seen.
        \item At each bucket boundary $b$, \alert{delete every entry with
              $\text{count}+\Delta \le b$} (too infrequent to matter).
      \end{itemize}
      \vspace{0.4em}
      On skewed data the true heavy hitters appear in the very first bucket and
      are \emph{never} pruned, so they end up counted essentially exactly.
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Guarantee}
        Deterministic: $\hat f(x)\le f(x)\le \hat f(x)+\varepsilon N$.
        No hashing, no randomness.
      \end{block}
      \begin{exampleblock}{Why it shines here}
        Near-zero error on the real Wikimedia stream: the top pages enter early
        and survive every prune.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Error structure: the key to everything}
  \centering
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{lll}
    \toprule
    \textbf{Algorithm} & \textbf{Error type} & \textbf{Space} \\
    \midrule
    Count--Min           & $f(x)\le \hat f(x)\le f(x)+\varepsilon N$ (one-sided $+$) & $\bigO{\tfrac{1}{\varepsilon}\log\tfrac1\delta}$ \\
    Count Sketch/FastAMS & $|\hat f(x)-f(x)|\le \varepsilon\lVert f\rVert_2$ (signed) & $\bigO{\tfrac{1}{\varepsilon^2}\log\tfrac1\delta}$ \\
    Misra--Gries         & $\hat f(x)\le f(x)\le \hat f(x)+\varepsilon N$ (one-sided $-$) & $\bigO{1/\varepsilon}$ \\
    Space-Saving         & over-estimate, isomorphic to MG & $\bigO{1/\varepsilon}$ \\
    Lossy Counting       & $\hat f(x)\le f(x)\le \hat f(x)+\varepsilon N$ & $\bigO{\tfrac1\varepsilon\log \varepsilon N}$ \\
    \bottomrule
  \end{tabular}

  \vspace{0.8em}
  \begin{itemize}
    \item CMS error scales with the $L_1$ mass $N$; FastAMS with the $L_2$ mass
          $\lVert f\rVert_2 \ll N$ on skewed data.
    \item \alert{One-sidedness matters:} CMS never under-counts (protects recall),
          MG never over-counts (protects precision).
  \end{itemize}
\end{frame}

\section{Dataset \& Streaming Model}
%================================================

\begin{frame}{Wikimedia pageviews \cite{wikimediaPageviews}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Hourly dumps $\to$ Parquet \texttt{(key, views)} pairs.
      \begin{itemize}
        \item \textbf{10{,}897{,}474} distinct \texttt{(project, page)} rows
        \item Total weight $N = 44{,}367{,}267$ views
        \item Strongly Zipfian, $\lVert f\rVert_2 \approx 5.5\times10^6$
        \item At $\varphi=10^{-4}$: threshold $\varphi N = 4{,}437$ views
              $\Rightarrow$ \textbf{158 true \hh{}s}
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Streaming model}
        Each row replayed as one \emph{weighted} element in scan order
        (\texttt{toLocalIterator}), a single-pass \textbf{cash-register}
        stream. Weighted single-window, not event-by-event replay.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{exampleblock}{Synthetic Zipfian streams (controlled skew)}
    Exponent $s\in\{0.6,0.8,1.0,1.2,1.5\}$ over a fixed vocabulary with
    deterministic keys $\Rightarrow$ the true top-$k$ is known exactly.
    Lower $s$ = flatter; higher $s$ = more concentrated.
  \end{exampleblock}
\end{frame}

\section{Experimental Design}
%================================================

\begin{frame}{Experimental design}
  \textbf{Platform:} Apache Spark \cite{zahariaApacheSparkUnified2016} in
  \texttt{local[*]} mode (all cores as parallel workers).
  \begin{block}{Five benchmark suites}
    \begin{enumerate}
      \item \textbf{Baseline detection}: literature-tuned params, real stream, $\varphi=10^{-4}$.
      \item \textbf{Same-memory sweep}: identical budgets (128\,KB--8\,MB).
      \item \textbf{Same-threshold sweep}: equal budgets under strict $\varphi N$.
      \item \textbf{Skew robustness}: fixed 1\,MB, fixed top-$k$, Zipf sweep + real snapshot.
      \item \textbf{Distributed partitioning \& mergeability}: vary $P\in\{1,2,4,8,16\}$.
    \end{enumerate}
  \end{block}
  \textbf{Metrics:} precision, recall, mean/max relative error, throughput
  (rows/s), summary memory, and \emph{merge fidelity} (max deviation vs.\ $P{=}1$).
  Ground truth: Spark \texttt{groupBy(key).sum(views)}.
\end{frame}

\section{Discussion}
%================================================

\begin{frame}{Which family, and when?}
  \begin{block}{Constraint-based recommendations}
    \begin{description}
      \item[\textcolor{sketchcol}{Lots of memory + no-false-negative guarantee}]
        \textbf{Count--Min}: cheap updates, top throughput, 100\% recall.
        (Error $\propto N$, so precision is expensive.)
      \item[Tight memory budget]
        \textbf{FastAMS} (tiny $L_2$ error on skew) or \textbf{Lossy Counting}
        (exact counting of the few real HHs). Need strict \emph{no false positives}?
        $\to$ \textbf{Misra--Gries}, at the cost of recall.
      \item[\textcolor{sketchcol}{Distributed / mergeable}]
        \textbf{Linear sketches}: embarrassingly parallel, \emph{lossless} merge.
        Counter summaries merge only approximately and drift with $P$.
      \item[Throughput-driven, accuracy-tolerant]
        \textbf{Sampling}: fastest, but only acceptable on highly skewed data.
    \end{description}
  \end{block}
  \centering\small
  \emph{Sketches win when memory is scarce or lossless merging matters;
  counters win when a one-sided guarantee and interpretable state are required.}
\end{frame}

\begin{frame}{Threats to validity}
  \begin{itemize}
    \item \textbf{Streaming model.} Weighted single-window stream in Parquet scan
          order, faithful for HH detection, but with no temporal drift or windowed HHs.
    \item \textbf{Local Spark.} \texttt{local[*]} reproduces merge correctness and
          relative scaling, but not HDFS I/O, network shuffle, or executor GC isolation.
          \emph{Relative} comparisons and the lossless-vs-lossy distinction carry over.
    \item \textbf{Detection-rule differences.} Counter summaries use a recall-preserving
          $(\varphi-\varepsilon)N$ report threshold in one suite and strict $\varphi N$ in
          another; comparable \emph{within} a suite, not across suites.
  \end{itemize}
\end{frame}

%================================================
\section{Conclusion}
%================================================

\begin{frame}{Conclusion}
  \begin{itemize}
    \item Implemented \& evaluated sketch-based, counter-based, sampling, and exact
          baselines on real Wikimedia + synthetic Zipfian streams across five suites.
    \item \textbf{At equal memory:} FastAMS and Lossy Counting are most accurate.
    \item \textbf{One-sided guarantees:} CMS trades memory for precision (keeps recall);
          MG trades recall for precision.
    \item \textbf{Distributed:} linear sketches merge losslessly, ideal for partitioned Spark;
          counters merge only approximately.
    \item \textbf{Convergence:} families agree as skew increases.
  \end{itemize}
  \vspace{0.5em}
  \begin{alertblock}{Take-away}
    There is no single ``best'' algorithm; the right choice is dictated by the
    \textbf{memory budget}, the required \textbf{error guarantee}, and the
    \textbf{distribution / system model}.
  \end{alertblock}
\end{frame}
